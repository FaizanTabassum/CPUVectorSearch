cmake_minimum_required(VERSION 3.16)
project(CPUVectorSearch VERSION 0.1.0 LANGUAGES CXX)

option(CVS_SANITIZERS "Enable address and undefined-behaviour sanitizers" OFF)
add_library(cpu_vector_search src/search.cpp src/io.cpp src/optimized_flat.cpp src/ivf.cpp src/kmeans.cpp)
target_include_directories(cpu_vector_search PUBLIC include)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|i.86)$" AND
   CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_sources(cpu_vector_search PRIVATE src/optimized_flat_avx2.cpp)
  set_source_files_properties(src/optimized_flat_avx2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2")
  target_compile_definitions(cpu_vector_search PRIVATE CVS_HAS_AVX2_IMPL=1)
endif()
target_compile_features(cpu_vector_search PUBLIC cxx_std_17)
set_target_properties(cpu_vector_search PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  # Keep FP32 scalar accumulation reproducible within this kernel.
  target_compile_options(cpu_vector_search PUBLIC -ffp-contract=off)
  target_compile_options(cpu_vector_search PRIVATE -Wall -Wextra -Wpedantic)
  if(CVS_SANITIZERS)
    target_compile_options(cpu_vector_search PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cpu_vector_search PUBLIC -fsanitize=address,undefined)
  endif()
endif()

add_executable(vector_search apps/benchmark.cpp)
target_link_libraries(vector_search PRIVATE cpu_vector_search)

# B's isolated layout/SIMD ablation; the full experiment runner is a separate task.
add_executable(flat_ablation apps/flat_ablation.cpp)
target_link_libraries(flat_ablation PRIVATE cpu_vector_search)

add_executable(reference_search apps/ground_truth.cpp)
target_link_libraries(reference_search PRIVATE cpu_vector_search)

include(CTest)
if(BUILD_TESTING)
  add_executable(search_tests tests/search_tests.cpp)
  add_executable(optimized_flat_tests tests/optimized_flat_tests.cpp)
  target_link_libraries(optimized_flat_tests PRIVATE cpu_vector_search)
  add_test(NAME optimized_flat_correctness COMMAND optimized_flat_tests)
  target_link_libraries(search_tests PRIVATE cpu_vector_search)
  add_test(NAME search_correctness COMMAND search_tests)
  add_executable(ivf_tests tests/ivf_tests.cpp)
  target_link_libraries(ivf_tests PRIVATE cpu_vector_search)
  add_test(NAME ivf_correctness COMMAND ivf_tests)
  add_executable(kmeans_tests tests/kmeans_tests.cpp)
  target_link_libraries(kmeans_tests PRIVATE cpu_vector_search)
  add_test(NAME kmeans_training COMMAND kmeans_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME data_pipeline COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_prepare_data.py)
    add_test(NAME end_to_end COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_end_to_end.py $<TARGET_FILE:vector_search>)
    add_test(NAME reference_answers COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_reference.py $<TARGET_FILE:reference_search>)
  endif()
endif()
